<script>
$(document).ready(function () {

    $("#customerloginForm").on("submit", function (e) {
        e.preventDefault(); // Stop normal form submission
        // const url1 = $(this).attr("action");
        // alert(url1);
        $.ajax({
            url: $(this).attr("action"),
            type: "POST",
            data: $(this).serialize(),
            dataType: "json",
            beforeSend: function () {
                $("#errorDiv").html('<div class="alert alert-info">Please wait...</div>');
            },
            success: function (res) {
                if (res.status == 1) {
                    $("#errorDiv").html(
                        '<div class="alert alert-success">' + res.message + '</div>'
                    );

                    // ✅ Redirect after success
                    setTimeout(function () {
                        window.location.href = "https://millionairescollection.com/";
                    }, 1000);
                } else {
                    $("#errorDiv").html(
                        '<div class="alert alert-danger">' + res.message + '</div>'
                    );
                }
            },
            error: function () {
                $("#errorDiv").html('<div class="alert alert-danger">Server error.</div>');
            }
        });
    });

});
</script>

<script>
$(document).on('submit', '#customerRegisterForm', function(e) {
    e.preventDefault();

    var formData = $(this).serialize();

    $.ajax({
        url: "https://millionairescollection.com/submit-customer-registration-form",
        type: "POST",
        data: formData,
        dataType: "json",
        success: function(res) {
            toastr.clear();

            if(res.status == 1) {

                toastr.success(res.message);

                $("#customerRegisterForm")[0].reset();

                // Show login form after success
                setTimeout(function() {
                    $("#signupSection").addClass("d-none");
                    $("#loginSection").removeClass("d-none");
                    toastr.info("Please login to continue.");
                }, 2000);

            } else {
                toastr.error(res.message);
            }
        },
        error: function() {
            toastr.clear();
            toastr.error("Server error! Try again later.");
        }
    });
});

</script>

function nextStep() {

  // rotate array
  positions.unshift(positions.pop());

  bottles.forEach((bottle, index) => {

        this.period = parseInt(period, 10) || 2000;
        this.txt = '';
        this.tick();
        this.isDeleting = false;

        TxtType.prototype.tick = function() {
            var i = this.loopNum % this.toRotate.length;
            var fullTxt = this.toRotate[i];
    
            if (this.isDeleting) {
            this.txt = fullTxt.substring(0, this.txt.length - 1);

            this.txt = fullTxt.substring(0, this.txt.length + 1);

            this.isDeleting = false;

<script id="4n2kds">
let currentStep = 0;
const steps = document.querySelectorAll('.process-step');

function showStep(index) {
    steps.forEach(step => step.classList.remove('active'));
    steps[index].classList.add('active');
}

function nextStep() {
    currentStep++;
    if (currentStep >= steps.length) {
        currentStep = 0; // loop back
    }
    showStep(currentStep);
}

function prevStep() {
    currentStep--;
    if (currentStep < 0) {
        currentStep = steps.length - 1;
    }
    showStep(currentStep);
}
</script>
